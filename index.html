// ══════════════════════════════════════════════════════════════ // CENTRO VIDA NUEVA — Apps Script API // Conecta Google Sheets con la web app del planificador // ══════════════════════════════════════════════════════════════ // ── NOMBRES DE HOJAS ────────────────────────────────────────── const HOJAS = { PERSONAS: "Personas", TAREAS: "Tareas", FRANJAS: "Franjas", CAPACIDADES: "Capacidades", DISPONIBILIDAD: "Disponibilidad", PLANTILLA: "Plantilla_semanal", PLANNING: "Planning", AUSENCIAS: "Ausencias", EXCEPCIONES: "Excepciones_Planning", HISTORIAL: "Historial", REGLAS: "Reglas" }; // ══════════════════════════════════════════════════════════════ // PUNTO DE ENTRADA — GET // La web app llama a esta URL con ?action=XXX // ══════════════════════════════════════════════════════════════ function doGet(e) { try { const action = e.parameter.action || "getDatos"; let resultado; switch(action) { case "getDatos": resultado = getDatos(); break; case "getPlanning": resultado = getPlanning(); break; default: resultado = { error: "Acción desconocida: " + action }; } return respuestaJSON(resultado); } catch(err) { return respuestaJSON({ error: err.message }); } } // ══════════════════════════════════════════════════════════════ // PUNTO DE ENTRADA — POST // La web app envía cambios con JSON en el body // ══════════════════════════════════════════════════════════════ function doPost(e) { try { const body = JSON.parse(e.postData.contents); const action = body.action; let resultado; switch(action) { case "guardarPlanning": resultado = guardarPlanning(body.planning, body.fecha); break; case "guardarPersona": resultado = guardarPersona(body.persona); break; case "eliminarPersona": resultado = eliminarPersona(body.personaID); break; case "guardarCapacidad": resultado = guardarCapacidad(body.capacidad); break; case "eliminarCapacidad": resultado = eliminarCapacidad(body.capacidadID); break; case "guardarDisponibilidad": resultado = guardarDisponibilidad(body.disponibilidad); break; case "guardarExcepcion": resultado = guardarExcepcion(body.excepcion); break; case "eliminarExcepcion": resultado = eliminarExcepcion(body.excepcionID); break; default: resultado = { error: "Acción desconocida: " + action }; } return respuestaJSON(resultado); } catch(err) { return respuestaJSON({ error: err.message }); } } // ══════════════════════════════════════════════════════════════ // LECTURA — Carga todos los datos de una vez // ══════════════════════════════════════════════════════════════ function getDatos() { return { personas: leerTabla(HOJAS.PERSONAS), tareas: leerTabla(HOJAS.TAREAS), franjas: leerTabla(HOJAS.FRANJAS), capacidades: leerTabla(HOJAS.CAPACIDADES), disponibilidad: leerTabla(HOJAS.DISPONIBILIDAD), plantilla: leerTabla(HOJAS.PLANTILLA), ausencias: leerTabla(HOJAS.AUSENCIAS), excepciones: leerExcepciones(), reglas: leerTabla(HOJAS.REGLAS) }; } function getPlanning() { return { planning: leerTabla(HOJAS.PLANNING) }; } // ══════════════════════════════════════════════════════════════ // ESCRITURA — Planning // ══════════════════════════════════════════════════════════════ function guardarPlanning(planning, fecha) { if(!planning || !planning.length) return { ok: false, error: "Sin datos de planning" }; const hoja = getHoja(HOJAS.PLANNING); const cabeceras = getCabeceras(hoja); // Eliminar filas de la misma semana const datos = hoja.getDataRange().getValues(); const fechaBase = new Date(fecha); const lunes = getLunes(fechaBase); const viernes = new Date(lunes); viernes.setDate(lunes.getDate() + 4); // Borrar desde la última fila hacia arriba para no desplazar índices for(let i = datos.length - 1; i >= 1; i--) { const fila = datos[i]; const idxFecha = cabeceras.indexOf("Fecha"); if(idxFecha === -1) continue; const fechaFila = new Date(fila[idxFecha]); fechaFila.setHours(0,0,0,0); const l = new Date(lunes); l.setHours(0,0,0,0); const v = new Date(viernes); v.setHours(23,59,59,0); if(fechaFila >= l && fechaFila <= v) { hoja.deleteRow(i + 1); } } // Insertar las nuevas filas planning.forEach(fila => { const nuevaFila = cabeceras.map(cab => fila[cab] !== undefined ? fila[cab] : ""); hoja.appendRow(nuevaFila); }); return { ok: true, insertadas: planning.length }; } // ══════════════════════════════════════════════════════════════ // ESCRITURA — Personas // ══════════════════════════════════════════════════════════════ function guardarPersona(persona) { if(!persona) return { ok: false, error: "Sin datos" }; const hoja = getHoja(HOJAS.PERSONAS); const cabeceras = getCabeceras(hoja); const datos = hoja.getDataRange().getValues(); const idxID = cabeceras.indexOf("PersonaID"); // Buscar si existe for(let i = 1; i < datos.length; i++) { if(datos[i][idxID] === persona.PersonaID) { // Actualizar fila existente const nuevaFila = cabeceras.map(cab => persona[cab] !== undefined ? persona[cab] : datos[i][cabeceras.indexOf(cab)]); hoja.getRange(i + 1, 1, 1, cabeceras.length).setValues([nuevaFila]); return { ok: true, accion: "actualizada" }; } } // Insertar nueva const nuevaFila = cabeceras.map(cab => persona[cab] !== undefined ? persona[cab] : ""); hoja.appendRow(nuevaFila); return { ok: true, accion: "creada" }; } function eliminarPersona(personaID) { if(!personaID) return { ok: false, error: "Sin PersonaID" }; eliminarFilasPorID(HOJAS.PERSONAS, "PersonaID", personaID); eliminarFilasPorID(HOJAS.CAPACIDADES, "PersonaID", personaID); eliminarFilasPorID(HOJAS.DISPONIBILIDAD, "PersonaID", personaID); eliminarFilasPorID(HOJAS.EXCEPCIONES, "PersonaID", personaID); return { ok: true }; } // ══════════════════════════════════════════════════════════════ // ESCRITURA — Capacidades // ══════════════════════════════════════════════════════════════ function guardarCapacidad(cap) { if(!cap) return { ok: false, error: "Sin datos" }; const hoja = getHoja(HOJAS.CAPACIDADES); const cabeceras = getCabeceras(hoja); const nuevaFila = cabeceras.map(cab => cap[cab] !== undefined ? cap[cab] : ""); hoja.appendRow(nuevaFila); return { ok: true }; } function eliminarCapacidad(capID) { if(!capID) return { ok: false, error: "Sin ID" }; // Buscar por CapacidadID o id const hoja = getHoja(HOJAS.CAPACIDADES); const cabeceras = getCabeceras(hoja); const datos = hoja.getDataRange().getValues(); const idxID = cabeceras.indexOf("CapacidadID") !== -1 ? cabeceras.indexOf("CapacidadID") : cabeceras.indexOf("id"); for(let i = datos.length - 1; i >= 1; i--) { if(String(datos[i][idxID]) === String(capID)) { hoja.deleteRow(i + 1); } } return { ok: true }; } // ══════════════════════════════════════════════════════════════ // ESCRITURA — Disponibilidad // ══════════════════════════════════════════════════════════════ function guardarDisponibilidad(registros) { if(!registros || !registros.length) return { ok: false, error: "Sin datos" }; const hoja = getHoja(HOJAS.DISPONIBILIDAD); const cabeceras = getCabeceras(hoja); const datos = hoja.getDataRange().getValues(); const idxPID = cabeceras.indexOf("PersonaID"); const idxDia = cabeceras.indexOf("Dia semana"); registros.forEach(reg => { let filaEncontrada = -1; for(let i = 1; i < datos.length; i++) { if(String(datos[i][idxPID]) === String(reg.PersonaID) && String(datos[i][idxDia]).toLowerCase() === String(reg["Dia semana"]).toLowerCase()) { filaEncontrada = i + 1; break; } } const nuevaFila = cabeceras.map(cab => reg[cab] !== undefined ? reg[cab] : ""); if(filaEncontrada !== -1) { hoja.getRange(filaEncontrada, 1, 1, cabeceras.length).setValues([nuevaFila]); } else { hoja.appendRow(nuevaFila); } }); return { ok: true }; } // ══════════════════════════════════════════════════════════════ // ESCRITURA — Excepciones // ══════════════════════════════════════════════════════════════ function guardarExcepcion(exc) { if(!exc) return { ok: false, error: "Sin datos" }; const hoja = getHoja(HOJAS.EXCEPCIONES); const cabeceras = getCabeceras(hoja); // Serializar franjas como JSON string si es array const excSer = {...exc}; if(Array.isArray(excSer.franjas)) excSer.franjas = JSON.stringify(excSer.franjas); const nuevaFila = cabeceras.map(cab => excSer[cab] !== undefined ? excSer[cab] : ""); hoja.appendRow(nuevaFila); return { ok: true }; } function eliminarExcepcion(excID) { if(!excID) return { ok: false, error: "Sin ID" }; const hoja = getHoja(HOJAS.EXCEPCIONES); const cabeceras = getCabeceras(hoja); const datos = hoja.getDataRange().getValues(); const idxID = cabeceras.findIndex(c => c.toLowerCase().includes("id")); for(let i = datos.length - 1; i >= 1; i--) { if(String(datos[i][idxID]) === String(excID)) { hoja.deleteRow(i + 1); } } return { ok: true }; } // ══════════════════════════════════════════════════════════════ // UTILIDADES // ══════════════════════════════════════════════════════════════ function leerTabla(nombreHoja) { try { const hoja = getHoja(nombreHoja); const lastRow = hoja.getLastRow(); const lastCol = hoja.getLastColumn(); if(lastRow < 2) return []; const rango = hoja.getRange(1, 1, lastRow, lastCol); const valores = rango.getValues(); const cabeceras = valores[0].map(c => String(c).trim()); return valores.slice(1) .filter(fila => fila.some(v => v !== "")) .map(fila => { const obj = {}; cabeceras.forEach((cab, i) => { let val = fila[i]; if(val instanceof Date) { const cabLower = cab.toLowerCase(); const esHora = cabLower.includes("hora") || cabLower.includes("inicio") || cabLower.includes("fin"); if(esHora) { // Formatear siempre como HH:mm con cero inicial val = Utilities.formatDate(val, "UTC", "HH:mm"); } else { val = Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd"); } } else if(val === true || val === "TRUE") { val = true; } else if(val === false || val === "FALSE") { val = false; } obj[cab] = val; }); return obj; }); } catch(e) { Logger.log("Error leyendo " + nombreHoja + ": " + e.message); return []; } } function leerExcepciones() { const registros = leerTabla(HOJAS.EXCEPCIONES); // Deserializar franjas si están como JSON string return registros.map(r => { if(r.franjas && typeof r.franjas === "string" && r.franjas.startsWith("[")) { try { r.franjas = JSON.parse(r.franjas); } catch(e) {} } else if(!r.franjas || r.franjas === "") { r.franjas = null; } return r; }); } function getHoja(nombre) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const hoja = ss.getSheetByName(nombre); if(!hoja) throw new Error("No existe la hoja: " + nombre); return hoja; } function getCabeceras(hoja) { return hoja.getRange(1, 1, 1, hoja.getLastColumn()).getValues()[0].map(c => String(c).trim()); } function eliminarFilasPorID(nombreHoja, columnaID, valor) { try { const hoja = getHoja(nombreHoja); const cabeceras = getCabeceras(hoja); const datos = hoja.getDataRange().getValues(); const idx = cabeceras.indexOf(columnaID); if(idx === -1) return; for(let i = datos.length - 1; i >= 1; i--) { if(String(datos[i][idx]) === String(valor)) hoja.deleteRow(i + 1); } } catch(e) { Logger.log("Error eliminando en " + nombreHoja + ": " + e.message); } } function getLunes(fecha) { const d = new Date(fecha); const dia = d.getDay(); d.setDate(d.getDate() - (dia === 0 ? 6 : dia - 1)); d.setHours(0,0,0,0); return d; } function respuestaJSON(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ══════════════════════════════════════════════════════════════ // TEST — ejecuta esto para probar que todo funciona // ══════════════════════════════════════════════════════════════ function testGetDatos() { const datos = getDatos(); Logger.log("Personas: " + datos.personas.length); Logger.log("Tareas: " + datos.tareas.length); Logger.log("Franjas: " + datos.franjas.length); Logger.log("Capacidades: " + datos.capacidades.length); Logger.log("Disponibilidad: " + datos.disponibilidad.length); Logger.log("Plantilla: " + datos.plantilla.length); Logger.log("Excepciones: " + datos.excepciones.length); Logger.log("Reglas: " + datos.reglas.length); }
